          avg_latency: (2.3 + (i * 0.5) + (j * 0.3)).toFixed(1) + 'ms',
        })
      }
      rows.push(baseRow)
    }
    cat[`bja-dsi-k8s-cluster-${pad}`] = { code, desc, data: rows }
  }
  return cat
}
